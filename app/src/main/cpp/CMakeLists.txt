cmake_minimum_required(VERSION 3.22.1)
project(livecaption_translation LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU ON CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_IS_DEV OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMIT "d81235049384534c167caea52b85a694f6103d14" CACHE STRING "" FORCE)
set(LLAMA_BUILD_NUMBER 0 CACHE STRING "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)

# Debug APKs are delivered for testing; inference still needs optimized kernels.
add_compile_options("$<$<CONFIG:Debug>:-O3>")

if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/llama.cpp/include/llama.h")
    message(FATAL_ERROR "Run python scripts/prepare_native_translation.py before building")
endif()
add_subdirectory(third_party/llama.cpp)
add_library(livecaption_translation SHARED translation_engine.cpp translation_jni.cpp)
target_link_libraries(livecaption_translation PRIVATE llama)
target_link_options(livecaption_translation PRIVATE "-Wl,-z,max-page-size=16384")
